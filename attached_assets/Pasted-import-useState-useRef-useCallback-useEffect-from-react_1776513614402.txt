// ─── ffmpeg.wasm loader ────────────────────────────────────────────────────────
let ffmpegInstance = null;

async function getFFmpeg() {
  if (ffmpegInstance) return ffmpegInstance;
  const { createFFmpeg, fetchFile } = await import("https://cdn.jsdelivr.net/npm/@ffmpeg/ffmpeg@0.11.6/dist/ffmpeg.min.js");
  const ff = createFFmpeg({ log: false });
  await ff.load();
  ffmpegInstance = { ff, fetchFile };
  return ffmpegInstance;
}

// ─── Convert any video/audio file to WAV via ffmpeg.wasm ──────────────────────
async function convertToWav(file, onProgress) {
  onProgress("Loading ffmpeg engine…");
  const { ff, fetchFile } = await getFFmpeg();
  onProgress("Converting to WAV…");
  const inputName = "input" + file.name.slice(file.name.lastIndexOf("."));
  ff.FS("writeFile", inputName, await fetchFile(file));
  await ff.run(
    "-i", inputName,
    "-vn",             // strip video track
    "-acodec", "pcm_s16le",
    "-ar", "44100",
    "-ac", "1",        // mono
    "output.wav"
  );
  const data = ff.FS("readFile", "output.wav");
  // cleanup
  try { ff.FS("unlink", inputName); } catch {}
  try { ff.FS("unlink", "output.wav"); } catch {}
  return new Blob([data.buffer], { type: "audio/wav" });
}

// ─── Audio utilities ───────────────────────────────────────────────────────────
function encodeWAV(samples, sampleRate) {
  const buf = new ArrayBuffer(44 + samples.length * 2);
  const v = new DataView(buf);
  const ws = (o, s) => { for (let i = 0; i < s.length; i++) v.setUint8(o + i, s.charCodeAt(i)); };
  ws(0, "RIFF"); v.setUint32(4, 36 + samples.length * 2, true);
  ws(8, "WAVE"); ws(12, "fmt ");
  v.setUint32(16, 16, true); v.setUint16(20, 1, true); v.setUint16(22, 1, true);
  v.setUint32(24, sampleRate, true); v.setUint32(28, sampleRate * 2, true);
  v.setUint16(32, 2, true); v.setUint16(34, 16, true);
  ws(36, "data"); v.setUint32(40, samples.length * 2, true);
  let off = 44;
  for (let i = 0; i < samples.length; i++) {
    const s = Math.max(-1, Math.min(1, samples[i]));
    v.setInt16(off, s < 0 ? s * 0x8000 : s * 0x7fff, true);
    off += 2;
  }
  return buf;
}

function extractSnippet(audioBuffer, offsetSec, durationSec = 10) {
  const sr = audioBuffer.sampleRate;
  const start = Math.floor(offsetSec * sr);
  const end = Math.min(start + Math.floor(durationSec * sr), audioBuffer.length);
  const len = end - start;
  if (len <= 0) return null;
  const mono = new Float32Array(len);
  for (let c = 0; c < audioBuffer.numberOfChannels; c++) {
    const ch = audioBuffer.getChannelData(c);
    for (let i = 0; i < len; i++) mono[i] += ch[start + i] / audioBuffer.numberOfChannels;
  }
  return new Blob([encodeWAV(mono, sr)], { type: "audio/wav" });
}

// ─── ACRCloud scan ─────────────────────────────────────────────────────────────
async function signRequest(str, secret) {
  const enc = new TextEncoder();
  const key = await crypto.subtle.importKey("raw", enc.encode(secret), { name: "HMAC", hash: "SHA-1" }, false, ["sign"]);
  const sig = await crypto.subtle.sign("HMAC", key, enc.encode(str));
  return btoa(String.fromCharCode(...new Uint8Array(sig)));
}

async function scanBlob(blob, winIdx) {
  const ts = Math.floor(Date.now() / 1000);
  const sig = await signRequest(["POST", "/v1/identify", ACR_ACCESS_KEY, "audio", "1", ts].join("\n"), ACR_ACCESS_SECRET);
  const fd = new FormData();
  fd.append("sample", blob, "sample.wav");
  fd.append("access_key", ACR_ACCESS_KEY);
  fd.append("data_type", "audio");
  fd.append("signature_version", "1");
  fd.append("signature", sig);
  fd.append("timestamp", ts);
  const res = await fetch(`https://${ACR_HOST}/v1/identify`, { method: "POST", body: fd });
  if (!res.ok) throw new Error(`ACRCloud HTTP ${res.status}`);
  const data = await res.json();
  console.log(`[CC] Window ${winIdx}:`, data?.status?.msg, data?.metadata?.music?.[0]?.title || "—");
  return data;
}

const isVideo = (f) => f.type.startsWith("video/") || [".mp4", ".mov", ".avi", ".mkv", ".webm"].some(e => f.name.toLowerCase().endsWith(e));

const STEP = { IDLE: "idle", CONVERTING: "converting", LOADING: "loading", READY: "ready", SCANNING: "scanning", DONE: "done", ERROR: "error" };

// ─── Component ────────────────────────────────────────────────────────────────
export default function ClipCleaner() {
  const [file, setFile] = useState(null);
  const [fileType, setFileType] = useState(""); // "video" | "audio"
  const [audioBuf, setAudioBuf] = useState(null);
  const [duration, setDuration] = useState(0);
  const [step, setStep] = useState(STEP.IDLE);
  const [statusMsg, setStatusMsg] = useState("");
  const [errorMsg, setErrorMsg] = useState("");
  const [progress, setProgress] = useState(0);
  const [curWin, setCurWin] = useState(0);
  const [totalWin, setTotalWin] = useState(0);
  const [results, setResults] = useState([]);
  const [dragOver, setDragOver] = useState(false);
  const inputRef = useRef();

  const reset = () => {
    setFile(null); setFileType(""); setAudioBuf(null); setDuration(0);
    setStep(STEP.IDLE); setStatusMsg(""); setErrorMsg("");
    setProgress(0); setCurWin(0); setTotalWin(0); setResults([]);
    ffmpegInstance = null; // reset ffmpeg so it reloads fresh
  };

  const processFile = useCallback(async (f) => {
    reset();
    setFile(f);

    // Basic validation
    const ext = "." + f.name.split(".").pop().toLowerCase();
    if (!SUPPORTED_EXTS.includes(ext)) {
      setErrorMsg(`"${f.name}" is not a supported format.\n\nSupported: MP4, MOV, AVI, MKV, WebM (video) · MP3, WAV, M4A, AAC (audio)`);
      setStep(STEP.ERROR);
      return;
    }
    if (f.size > MAX_SIZE_MB * 1024 * 1024) {
      setErrorMsg(`File too large (${(f.size / 1024 / 1024).toFixed(0)}MB). Maximum is ${MAX_SIZE_MB}MB.`);
      setStep(STEP.ERROR);
      return;
    }

    const video = isVideo(f);
    setFileType(video ? "video" : "audio");
    console.log(`[CC] File: ${f.name} | Type: ${video ? "video" : "audio"} | Size: ${(f.size / 1024 / 1024).toFixed(1)}MB`);

    let wavBlob;

    if (video) {
      // ── VIDEO PATH: extract audio via ffmpeg.wasm ──
      setStep(STEP.CONVERTING);
      try {
        wavBlob = await convertToWav(f, (msg) => setStatusMsg(msg));
        console.log(`[CC] Converted to WAV: ${(wavBlob.size / 1024).toFixed(0)}KB`);
      } catch (e) {
        console.error("[CC] ffmpeg conversion failed:", e);
        setErrorMsg(
          `Could not extract audio from "${f.name}".\n\n` +
          `Error: ${e.message}\n\n` +
          `Try converting to MP3 first using VLC or cloudconvert.com.`
        );
        setStep(STEP.ERROR);
        return;
      }
    } else {
      // ── AUDIO PATH: try direct decode, fallback to ffmpeg ──
      setStep(STEP.LOADING);
      const ab = await f.arrayBuffer();
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      const ctx = new AudioCtx();
      try {
        const decoded = await ctx.decodeAudioData(ab);
        setAudioBuf(decoded);
        setDuration(decoded.duration);
        console.log(`[CC] Direct decode OK: ${decoded.duration.toFixed(1)}s`);
        setStep(STEP.READY);
        return;
      } catch {
        ctx.close();
        console.warn("[CC] Direct decode failed, falling back to ffmpeg…");
        setStep(STEP.CONVERTING);
        setStatusMsg("Converting audio format…");
        try {
          wavBlob = await convertToWav(f, (msg) => setStatusMsg(msg));
        } catch (e) {
          setErrorMsg(`Could not decode "${f.name}".\n\nPlease convert to MP3 or WAV first.\n\nFree converter: cloudconvert.com`);
          setStep(STEP.ERROR);
          return;
        }
      }
    }

    // ── Decode the WAV blob via Web Audio API ──
    setStep(STEP.LOADING);
    setStatusMsg("Decoding audio…");
    try {
      const ab = await wavBlob.arrayBuffer();
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      const ctx = new AudioCtx();
      const decoded = await ctx.decodeAudioData(ab);
      console.log(`[CC] WAV decode OK: ${decoded.duration.toFixed(1)}s, ${decoded.sampleRate}Hz`);
      setAudioBuf(decoded);
      setDuration(decoded.duration);
      setStep(STEP.READY);
    } catch (e) {
      console.error("[CC] WAV decode failed:", e);
      setErrorMsg("Failed to decode the converted audio. Please try a different file.");
      setStep(STEP.ERROR);
    }
  }, []);

  const startScan = async () => {
    if (!audioBuf) return;
    setStep(STEP.SCANNING);
    setResults([]);
    setProgress(0);

    const steps = Math.max(1, Math.ceil(duration / 30));
    setTotalWin(steps);
    const found = [];

    console.log(`[CC] Scan start: ${steps} windows`);

    for (let i = 0; i < steps; i++) {
      setCurWin(i + 1);
      setProgress(Math.round((i / steps) * 100));
      const blob = extractSnippet(audioBuf, i * 30, 10);
      if (!blob) continue;
      try {
        const data = await scanBlob(blob, i + 1);
        if (data?.status?.code === 0 && data?.metadata?.music?.length > 0) {
          const t = data.metadata.music[0];
          found.push({
            offset: i * 30,
            end: Math.min((i + 1) * 30, duration),
            title: t.title || "Unknown Title",
            artist: t.artists?.[0]?.name || "Unknown Artist",
            album: t.album?.name || "",
            label: t.label || "",
            score: t.score,
          });
        }
      } catch (e) {
        setErrorMsg(`Network error on window ${i + 1}: ${e.message}`);
        setStep(STEP.ERROR);
        return;
      }
    }

    console.log(`[CC] Scan done. ${found.length} match(es).`);
    setProgress(100);
    setResults(found);
    setStep(STEP.DONE);
  };

  // ─── Risk ──────────────────────────────────────────────────────────────────
  const risk = results.length === 0 ? "safe" : results.length <= 2 ? "medium" : "high";

  const riskIcon = { safe: "✅", medium: "⚠️", high: "🚨" }[risk];
  const riskLabel = { safe: "No copyright detected", medium: "Copyright risk found", high: "High copyright risk" }[risk];

  // ─── Shared styles ────────────────────────────────────────────────────────
  const card = { background: "#0d0d18", border: "1px solid #1a1a28", borderRadius: 12, padding: "16px 20px" };
  const btnPrimary = { width: "100%", padding: "15px", background: "linear-gradient(135deg,#00e5ff,#7c3aed)", border: "none", borderRadius: 12, color: "#000", fontWeight: 700, fontSize: 15, cursor: "pointer", fontFamily: "inherit" };
  const btnSecondary = { width: "100%", padding: "13px", background: "transparent", border: "1px solid #1e1e30", borderRadius: 12, color: "#555", fontWeight: 600, fontSize: 13, cursor: "pointer", fontFamily: "inherit" };

  return (
    <div style={{ minHeight: "100vh", background: "#0a0a0f", color: "#e8e8f0", fontFamily: "'Courier New', monospace", margin: 0, padding: 0 }}>

      {/* Header */}
      <div style={{ borderBottom: "1px solid #1a1a28", padding: "16px 24px", display: "flex", alignItems: "center", gap: 12, background: "#0d0d18" }}>
        <div style={{ width: 32, height: 32, borderRadius: 8, background: "linear-gradient(135deg,#00e5ff,#7c3aed)", display: "flex", alignItems: "center", justifyContent: "center", fontSize: 15, flexShrink: 0 }}>✂</div>
        <div>
          <div style={{ fontSize: 16, fontWeight: 700, color: "#fff" }}>ClipCleaner</div>
          <div style={{ fontSize: 10, color: "#444", letterSpacing: "1.5px", textTransform: "uppercase" }}>Copyright Pre-Check · Beta v3</div>
        </div>
        <div style={{ marginLeft: "auto", fontSize: 10, color: "#00e5ff", border: "1px solid #00e5ff22", padding: "3px 8px", borderRadius: 4 }}>
          Video + Audio
        </div>
      </div>

      <div style={{ maxWidth: 640, margin: "0 auto", padding: "32px 18px" }}>

        {/* Pipeline diagram */}
        {step === STEP.IDLE && (
          <div>
            <div style={{ ...card, marginBottom: 16, display: "flex", justifyContent: "space-around", alignItems: "center", flexWrap: "wrap", gap: 8 }}>
              {["Upload", "Extract Audio", "Convert to WAV", "Scan", "Results"].map((s, i, arr) => (
                <div key={s} style={{ display: "flex", alignItems: "center", gap: 6 }}>
                  <div style={{ fontSize: 11, color: "#3a3a5a", textAlign: "center" }}>{s}</div>
                  {i < arr.length - 1 && <div style={{ fontSize: 11, color: "#252535" }}>→</div>}
                </div>
              ))}
            </div>

            {/* Drop zone */}
            <div
              onDragOver={(e) => { e.preventDefault(); setDragOver(true); }}
              onDragLeave={() => setDragOver(false)}
              onDrop={(e) => { e.preventDefault(); setDragOver(false); const f = e.dataTransfer.files[0]; if (f) processFile(f); }}
              onClick={() => inputRef.current.click()}
              style={{
                border: `2px dashed ${dragOver ? "#00e5ff" : "#252535"}`,
                borderRadius: 14, padding: "52px 20px", textAlign: "center",
                cursor: "pointer", background: dragOver ? "rgba(0,229,255,0.04)" : "#0d0d18",
                transition: "all 0.2s",
              }}
            >
              <div style={{ fontSize: 40, marginBottom: 12 }}>🎬</div>
              <div style={{ fontSize: 15, fontWeight: 600, color: "#a0a0c0", marginBottom: 6 }}>Drop your video or audio file here</div>
              <div style={{ fontSize: 12, color: "#333", marginBottom: 18 }}>
                MP4 · MOV · AVI · MKV · WebM · MP3 · WAV · M4A · AAC
              </div>
              <div style={{ display: "inline-block", padding: "8px 20px", borderRadius: 8, background: "rgba(0,229,255,0.07)", border: "1px solid rgba(0,229,255,0.2)", fontSize: 12, color: "#00e5ff" }}>
                Browse files
              </div>

            </div>
            <div style={{ marginTop: 12, fontSize: 11, color: "#1e1e2e", textAlign: "center" }}>Max {MAX_SIZE_MB}MB · Audio extracted automatically from video files</div>
          </div>
        )}

        {/* CONVERTING */}
        {step === STEP.CONVERTING && (
          <div style={{ textAlign: "center", padding: "60px 20px" }}>
            <div style={{ fontSize: 32, marginBottom: 16 }}>⚙️</div>
            <div style={{ fontSize: 14, color: "#888", marginBottom: 6 }}>{statusMsg || "Processing…"}</div>

              This may take 10–30 seconds for large files

          </div>
        )}

        {/* LOADING */}
        {step === STEP.LOADING && (
          <div style={{ textAlign: "center", padding: "60px 20px" }}>
            <div style={{ fontSize: 32, marginBottom: 16 }}>⏳</div>
            <div style={{ fontSize: 13, color: "#555" }}>{statusMsg || `Decoding ${file?.name}…`}</div>
          </div>
        )}

        {/* ERROR */}
        {step === STEP.ERROR && (
          <div>
            <div style={{ background: "#140808", border: "1px solid #ff333322", borderRadius: 12, padding: "20px 22px", marginBottom: 16 }}>
              <div style={{ fontSize: 12, fontWeight: 700, color: "#ff6666", marginBottom: 8 }}>⚠ Error</div>
              <div style={{ fontSize: 12, color: "#cc6666", lineHeight: 1.8, whiteSpace: "pre-line" }}>{errorMsg}</div>
            </div>
            <div style={{ ...card, marginBottom: 16, fontSize: 12, color: "#3a3a5a", lineHeight: 1.8 }}>
              <div style={{ color: "#555", marginBottom: 6, fontWeight: 600 }}>Free converters:</div>
              <div>• cloudconvert.com — convert any format to MP3</div>
              <div>• online-audio-converter.com</div>
              <div>• VLC → Media → Convert/Save</div>
            </div>
            <button onClick={reset} style={btnSecondary}>↩ Try another file</button>

          <div>

              <div>
                <div style={{ fontSize: 13, fontWeight: 600, color: "#c8c8e0" }}>Scanning audio…</div>
